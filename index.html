① Apps Script の設定手順
1. Googleスプレッドシート を新規作成し、シート名を 打刻データ にする
2. メニュー「拡張機能」→「Apps Script」を開く
3. 既存のコードをすべて削除してから以下を貼り付けて保存（Ctrl+S）
function doGet(e){
  const sheet=SpreadsheetApp.getActiveSpreadsheet().getSheetByName('打刻データ');
  const p=e.parameter;
  if(p.action==='post'&&p.key&&p.field&&p.val){
    const found=sheet.createTextFinder(p.key).findNext();
    if(found){
      const col=p.field==='in'?2:3;
      sheet.getRange(found.getRow(),col).setValue(p.val);
    }else{
      const row=p.field==='in'?[p.key,p.val,'']:[ p.key,'',p.val];
      sheet.appendRow(row);
    }
    return ContentService.createTextOutput('ok');
  }
  const rows=sheet.getDataRange().getValues();
  const result={};
  rows.forEach(r=>{if(r[0])result[r[0]]={in:r[1]||'',out:r[2]||''}});
  const out=ContentService.createTextOutput(JSON.stringify(result));
  out.setMimeType(ContentService.MimeType.JSON);
  return out;
}
4. 「デプロイ」→「デプロイを管理」→ 鉛筆アイコン →「バージョン」を新バージョンに変更 →「デプロイ」
